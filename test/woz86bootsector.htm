Command Syntax

The monitor uses Woz Monitor-style syntax (all addresses in hexadecimal):

  • All addresses are 4-digit hexadecimal (0000-FFFF)
  • All byte values are 2-digit hexadecimal (00-FF)
  • Commands are case-insensitive (R/r, G/g work the same)
  • Press Enter/Return after each command
  • No spaces required (except between bytes in write command)

; 16-bit Serial Bootloader Monitor
; Woz Monitor style commands via COM1 serial port
; Fits in 512-byte boot sector

org 0x7C00
bits 16

start:
    ; Initialize segments
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00          ; Stack grows down from bootloader

    ; Initialize COM1 serial port (0x3F8)
    ; Disable interrupts
    mov dx, 0x3F9           ; Interrupt Enable Register
    xor al, al
    out dx, al

    ; Set DLAB (Divisor Latch Access Bit) to configure baud rate
    mov dx, 0x3FB           ; Line Control Register
    mov al, 0x80            ; DLAB = 1
    out dx, al

    ; Set baud rate to 9600 (divisor = 115200/9600 = 12)
    mov dx, 0x3F8           ; Divisor Latch Low
    mov al, 12
    out dx, al
    mov dx, 0x3F9           ; Divisor Latch High
    xor al, al
    out dx, al

    ; Configure: 8 data bits, no parity, 1 stop bit (8N1)
    mov dx, 0x3FB           ; Line Control Register
    mov al, 0x03            ; DLAB=0, 8N1
    out dx, al

    ; Enable DTR and RTS
    mov dx, 0x3FC           ; Modem Control Register
    mov al, 0x03            ; DTR + RTS
    out dx, al

    ; Fall through to main loop

Main Monitor Loop

main_loop:
    ; Print prompt
    mov al, 0x0D            ; CR
    call serial_send
    mov al, 0x0A            ; LF
    call serial_send
    mov al, '*'
    call serial_send
    mov al, ' '
    call serial_send

    ; Parse input and execute command
    call parse_input

    jmp main_loop

Serial I/O Primitives


; Send byte via serial
; Input: AL = byte to send
serial_send:
    push dx
    push ax
    mov dx, 0x3FD           ; Line Status Register
.wait:
    in al, dx
    test al, 0x20           ; Transmit buffer empty?
    jz .wait
    pop ax
    mov dx, 0x3F8           ; Data register
    out dx, al
    pop dx
    ret

; Receive byte via serial
; Output: AL = received byte
serial_recv:
    mov dx, 0x3FD           ; Line Status Register
.wait:
    in al, dx
    test al, 0x01           ; Data ready?
    jz .wait
    mov dx, 0x3F8           ; Data register
    in al, dx
    ret

; Print null-terminated string
; Input: SI = pointer to string
print_string:
    lodsb
    test al, al
    jz .done
    call serial_send
    jmp print_string
.done:
    ret

Hex Output Routines


; Print hex nibble (4 bits)
; Input: AL = nibble (0-15)
print_hex_nibble:
    and al, 0x0F
    add al, '0'
    cmp al, '9'
    jbe .send
    add al, 7               ; 'A' - '9' - 1 = 7
.send:
    jmp serial_send         ; Tail call

; Print hex byte (8 bits)
; Input: AL = byte
print_hex_byte:
    push ax
    shr al, 4               ; High nibble
    call print_hex_nibble
    pop ax
    and al, 0x0F            ; Low nibble
    jmp print_hex_nibble    ; Tail call

; Print hex word (16 bits)
; Input: AX = word
print_hex_word:
    xchg al, ah             ; Print high byte first
    call print_hex_byte
    xchg al, ah             ; Restore and print low byte
    jmp print_hex_byte      ; Tail call

Hex Input Parsing


; Parse hex nibble from input
; Output: AL = nibble value (0-15), or 0xFF if invalid
parse_hex_nibble:
    call serial_recv
    call serial_send        ; Echo character
    cmp al, '0'
    jb .invalid
    cmp al, '9'
    jbe .digit
    or al, 0x20             ; Convert to lowercase
    cmp al, 'a'
    jb .invalid
    cmp al, 'f'
    ja .invalid
    sub al, 'a' - 10
    ret
.digit:
    sub al, '0'
    ret
.invalid:
    mov al, 0xFF
    ret

; Parse hex number (up to 4 digits for 16-bit address)
; Output: BX = parsed address, AL = command character
parse_hex_number:
    xor bx, bx              ; Accumulator
    xor cx, cx              ; Digit counter
.loop:
    call serial_recv

    ; Echo character
    push ax
    call serial_send
    pop ax

    ; Check if hex digit
    cmp al, '0'
    jb .done
    cmp al, '9'
    jbe .digit
    or al, 0x20             ; Convert to lowercase
    cmp al, 'a'
    jb .done
    cmp al, 'f'
    ja .done
    sub al, 'a' - 10
    jmp .add_digit
.digit:
    sub al, '0'
.add_digit:
    ; Shift accumulator left 4 bits and add new nibble
    shl bx, 4
    or bl, al
    inc cx
    cmp cx, 4               ; Max 4 hex digits (16-bit address)
    jb .loop
    ; Read one more char (command char)
    call serial_recv
    push ax
    call serial_send
    pop ax
.done:
    ret                     ; BX = address, AL = command character

Command Parser and Dispatcher


; Parse input and dispatch command
; Commands:
;   AAAA       - Examine byte at address
;   AAAAR          - Dump memory from address
;   AAAA:BB BB     - Write bytes to address
;   AAAAG          - Execute at address
parse_input:
    call parse_hex_number   ; BX = address, AL = command char

    ; AL now contains the character after the address
    ; Check for CR (examine)
    cmp al, 0x0D
    je cmd_examine

    ; Check for 'R' or 'r' (dump range)
    cmp al, 'R'
    je cmd_dump_range
    cmp al, 'r'
    je cmd_dump_range

    ; Check for 'G' or 'g' (execute)
    cmp al, 'G'
    je cmd_execute
    cmp al, 'g'
    je cmd_execute

    ; Check for ':' (write bytes)
    cmp al, ':'
    je cmd_write_bytes

    ; Unknown command - skip to end of line
    jmp skip_to_eol

Commands


; Examine byte at address
; Input: BX = address
cmd_examine:
    mov ax, bx
    call print_hex_word

    mov al, ':'
    call serial_send
    mov al, ' '
    call serial_send

    mov al, [bx]
    call print_hex_byte
    ret

; Dump memory range (8 lines x 16 bytes = 128 bytes)
; Input: BX = start address
cmd_dump_range:
    mov di, bx
    mov cx, 8               ; 8 lines
.line_loop:
    push cx

    ; Newline
    mov al, 0x0D
    call serial_send
    mov al, 0x0A
    call serial_send

    ; Print address
    mov ax, di
    call print_hex_word
    mov al, ':'
    call serial_send
    mov al, ' '
    call serial_send

    ; Print 16 bytes
    mov cx, 16
.byte_loop:
    mov al, [di]
    inc di
    call print_hex_byte
    mov al, ' '
    call serial_send
    loop .byte_loop

    pop cx
    loop .line_loop
    ret

; Write bytes to memory
; Input: BX = start address
cmd_write_bytes:
    mov di, bx
.loop:
    ; Read character
    call serial_recv

    ; Echo it
    push ax
    call serial_send
    pop ax

    ; Skip spaces
    cmp al, ' '
    je .loop

    ; Check for CR (done)
    cmp al, 0x0D
    je .done

    ; Parse high nibble (AL already has the character)
    cmp al, '0'
    jb .skip_to_cr
    cmp al, '9'
    jbe .digit1
    or al, 0x20
    cmp al, 'a'
    jb .skip_to_cr
    cmp al, 'f'
    ja .skip_to_cr
    sub al, 'a' - 10
    jmp .got_high
.digit1:
    sub al, '0'
.got_high:
    shl al, 4
    mov ah, al

    ; Parse low nibble
    call serial_recv
    push ax
    call serial_send
    pop ax
    cmp al, '0'
    jb .skip_to_cr
    cmp al, '9'
    jbe .digit2
    or al, 0x20
    cmp al, 'a'
    jb .skip_to_cr
    cmp al, 'f'
    ja .skip_to_cr
    sub al, 'a' - 10
    jmp .got_low
.digit2:
    sub al, '0'
.got_low:
    or al, ah
    stosb                   ; Store byte and increment DI
    jmp .loop

.skip_to_cr:
    call serial_recv
    push ax
    call serial_send
    pop ax
    cmp al, 0x0D
    jne .skip_to_cr
.done:
    ret

; Execute code at address
; Input: BX = address
cmd_execute:
    call bx                 ; Call the address
    ret                     ; If it returns, continue

; Skip to end of line
skip_to_eol:
    call serial_recv
    cmp al, 0x0D
    jne skip_to_eol
    ret
